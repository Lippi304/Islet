<!-- Exact analog source, ClipboardStore.swift (31 lines, full file) — copy this shape, drop the
     firstIndex(where: kind ==) dedupe branch (append becomes a plain append), add remove(id:) -->
struct ClipboardStore: Equatable {
    private(set) var items: [ClipboardItem] = []
    let cap = 30

    mutating func append(_ item: ClipboardItem) {
        if let index = items.firstIndex(where: { $0.kind == item.kind }) {
            items.remove(at: index)
            items.append(item)
            return
        }
        items.append(item)
        if items.count > cap { items.removeFirst() }
    }

    mutating func clear() {
        items.removeAll()
    }
}
