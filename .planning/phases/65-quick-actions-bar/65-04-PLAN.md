```swift
@MainActor
final class FocusModeMonitor {
    static var isAuthorized: Bool {
        INFocusStatusCenter.default.authorizationStatus == .authorized
    }
    static func requestAuthorization(completion: @escaping (Bool) -> Void) {
        INFocusStatusCenter.default.requestAuthorization { status in
            completion(status == .authorized)
        }
    }
    private func poll() {
        guard INFocusStatusCenter.default.authorizationStatus == .authorized else { return }
        guard let isFocused = INFocusStatusCenter.default.focusStatus.isFocused else { return }
        onChange(isFocused)
    }
}
```
Pitfall 4 (RESEARCH.md): this codebase has TWO separate Focus-related features sharing the same
`INFocusStatusCenter` OS-level authorization bucket — Phase 38's `FocusActivity`/`activity.focus`
ambient HUD (DETECTS Focus state) and this phase's Quick Actions DND toggle (ATTEMPTS TO CHANGE
it). `FocusToggleAction` must gate its availability on `ActivitySettings.quickActionsKey` only,
NEVER on `ActivitySettings.focusKey` — these stay decoupled settings even though they share one
OS permission bucket.

RESEARCH.md Code Example / Pattern 2 (target shape, builds on FocusModeMonitor's exact entry
points):
```swift
enum FocusToggleAction {
    static func toggle(shortcutName: String, onResult: @escaping (Bool) -> Void) {
        guard FocusModeMonitor.isAuthorized else { onResult(false); return }
        let before = INFocusStatusCenter.default.focusStatus.isFocused ?? false
        let task = Process()
        task.executableURL = URL(fileURLWithPath: "/usr/bin/shortcuts")
        task.arguments = ["run", shortcutName]
        do { try task.run() } catch { onResult(false); return }
        task.terminationHandler = { _ in
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.5) {
                let after = INFocusStatusCenter.default.focusStatus.isFocused ?? before
                onResult(after != before)
            }
        }
    }
}
```
RESEARCH.md Open Question 1's explicit recommendation: treat "no Shortcut configured yet" as the
QACTION-03 failure state ITSELF (visible, not silent) rather than adding new setup-flow UI this
session — a fixed, documented Shortcut name is invoked; if it doesn't exist on the user's Mac,
`shortcuts run` fails/no-ops and the read-back naturally reports failure, which IS the required
"visible, not silent" behavior with zero extra config surface needed.
