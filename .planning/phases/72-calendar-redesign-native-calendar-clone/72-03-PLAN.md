private func dayEventsList(_ dayEvents: [EventInput]) -> some View {
    ScrollView(.vertical) {
        VStack(alignment: .leading, spacing: 8) {
            ForEach(Array(dayEvents.enumerated()), id: \.offset) { _, event in
                HStack(spacing: 6) {
                    Circle()
                        .fill(Color(red: event.colorRed, green: event.colorGreen, blue: event.colorBlue))
                        .frame(width: 6, height: 6)
                    Text(event.title)
                        .font(.system(size: 12, weight: .regular, design: .rounded))
                        .foregroundStyle(.white)
                        .lineLimit(1)
                        .truncationMode(.tail)
                    Spacer(minLength: 4)
                    Text(event.start, format: .dateTime.hour().minute())
                        .font(.system(size: 11, weight: .regular, design: .rounded))
                        .foregroundStyle(.secondary)
                        .monospacedDigit()
                }
                .padding(.horizontal, 12)
                .padding(.vertical, 8)
                .background(
                    RoundedRectangle(cornerRadius: 8, style: .continuous)
                        .fill(Color.white.opacity(0.06))
                )
            }
        }
    }
    .scrollIndicators(.never)
}

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-72-05 | Tampering (malformed display content) | `EventRow`'s `.help(event.title)` tooltip, `EventEditPopover`'s pre-filled `TextField` | mitigate | Both surfaces render `event.title` only as a plain `Text`/native tooltip or a bound `TextField` value — never interpolated into a format/log/shell string (T-14-06, reused verbatim); the existing `.lineLimit(1)/.truncationMode(.tail)` stays on the row's own title `Text`, the tooltip shows the full string but only via the native `.help()` API, never a custom rendering path |
| T-72-06 | Repudiation / accidental data loss | Hover-reveal delete with no confirm dialog (D-08) | accept | Explicit, locked user decision (D-08) — EventKit deletion is recoverable via Calendar.app/iCloud sync history, not a permanent unrecoverable action; a confirm dialog was deliberately rejected as unnecessary friction |
